P0 invoices.category_id FK: NO ACTION — törlés blokkolja a rendszert

A invoices tábla category_id mezőjének FK constraint-je NO ACTION szabályt használ törléskor. Ha egy kategóriát törölnek, az összes hozzákapcsolt invoice megakadályozza a törlést, FK violation hibát dobva.

-- Jelenlegi viselkedés:
-- invoices.category_id → categories.id  ON DELETE NO ACTION ❌

-- Javasolt fix:
ALTER TABLE public.invoices
  DROP CONSTRAINT invoices_project_id_fkey,  -- a constraint neve félrevezető!
  ADD CONSTRAINT invoices_category_id_fkey
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE SET NULL;
⚠️ FIGYELEM: A constraint neve invoices_project_id_fkey de a category_id oszlopra mutat! Ez egy migrációs hiba, ami nagyon nehezen debugolható.

Javítás:

DROP POLICY "Service role can manage invoice items" ON public.nav_invoice_items;

-- Hozd létre company_members alapú policy-t:
CREATE POLICY "Members can view nav_invoice_items" ON public.nav_invoice_items
  FOR SELECT USING (
    EXISTS (
      SELECT 1 FROM nav_invoices ni
      JOIN company_members cm ON cm.company_id = ni.company_id
      WHERE ni.id = nav_invoice_items.nav_invoice_id
        AND cm.user_id = auth.uid()
    )
  );

-- INSERT/UPDATE/DELETE-re service_role-t használj, ne public-ot

P0 Auto-approve side-effect a queryFn-ben

// TransactionsPage.tsx (L243-260):

  // ❌ WRITE MŰVELET egy READ QUERY-BEN!
  const toAutoApprove = fetchedTransactions.filter(
    t => t.matched_invoice_id && !t.is_verified && t.confidence_score >= 0.9
  );
  if (toAutoApprove.length > 0) {
    await supabase
      .from('transactions')
      .update({ is_verified: true, match_type: 'auto' })
      .in('id', ids);
  }
}

Miért kritikus:

  • Minden oldal-navigáció, refetch, vagy TanStack Query retry alkalmával mellékhatás történik
  • Ha a query hibázik a write után, a felhasználó azt látja hogy hiba van, de az adat már módosult
  • Tesztelhetetlen, nem szándékos duplikált futások

Javítás — DB trigger:

CREATE FUNCTION auto_approve_high_confidence() RETURNS trigger AS $$
BEGIN
  IF NEW.confidence_score >= 0.9 AND NEW.matched_invoice_id IS NOT NULL THEN
    NEW.is_verified := true;
    NEW.match_type := 'auto';
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

// lib/exportUtils.ts [NEW]
import * as XLSX from 'xlsx';
import { toast } from 'sonner';

export function exportToFile(
  headers: string[], data: string[][],
  format: 'csv' | 'xlsx', filename: string
) {
  const timestamp = new Date().toISOString().slice(0, 19).replace(/:/g, '-');
  if (format === 'csv') {
    const csvContent = [headers.join(','),
      ...data.map(row => row.map(cell => `"${cell}"`).join(','))
    ].join('\n');
    const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    link.href = URL.createObjectURL(blob);
    link.download = `${filename}_${timestamp}.csv`;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    toast.success('Exportálva CSV formátumban');
  } else {
    const ws = XLSX.utils.aoa_to_sheet([headers, ...data]);
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, filename);
    XLSX.writeFile(wb, `${filename}_${timestamp}.xlsx`);
    toast.success('Exportálva XLSX formátumban');
  }
}
